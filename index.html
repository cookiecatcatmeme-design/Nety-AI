from sklearn.model_selection import train_test_split from sklearn.ensemble import RandomForestClassifier from sklearn.datasets import load_iris # 1. Load sample dataset data = load_iris() X_train, X_test, y_train, y_test = train_test_split(data.data, data.target, test_size=0.2) # 2. Initialize the AI model model = RandomForestClassifier() # 3. Train the model on the data model.fit(X_train, y_train) # 4. Check accuracy accuracy = model.score(X_test, y_test) print(f"Model Accuracy: {accuracy * 100:.2f}%")
