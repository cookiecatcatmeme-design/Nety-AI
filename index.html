import torch import torch.nn as nn import torch.optim as optim # 1. Define the Neural Network Architecture class SimpleAI(nn.Module): def __init__(self, input_size, hidden_size, output_size): super(SimpleAI, self).__init__() # Input layer -> Hidden layer self.fc1 = nn.Linear(input_size, hidden_size) # Activation function (adds non-linear reasoning) self.relu = nn.ReLU() # Hidden layer -> Output layer self.fc2 = nn.Linear(hidden_size, output_size) def forward(self, x): x = self.fc1(x) x = self.relu(x) x = self.fc2(x) return x # 2. Create sample dummy data (5 samples, 4 features each) # Example: 4 features like height, weight, age, score -> target class 0 or 1 X_train = torch.randn(5, 4) y_train = torch.tensor([0, 1, 0, 1, 0]) # 3. Instantiate the Model model = SimpleAI(input_size=4, hidden_size=8, output_size=2) # 4. Set Loss Function and Optimizer criterion = nn.CrossEntropyLoss() optimizer = optim.Adam(model.parameters(), lr=0.01) # 5. Train the AI (100 training cycles/epochs) print("Training the AI model...") for epoch in range(100): # Forward Pass: Make predictions outputs = model(X_train) loss = criterion(outputs, y_train) # Backward Pass: Calculate errors and adjust weights optimizer.zero_grad() loss.backward() optimizer.step() if (epoch + 1) % 20 == 0: print(f"Epoch [{epoch + 1}/100], Loss: {loss.item():.4f}") # 6. Test the AI on a new unseen input test_data = torch.randn(1, 4) prediction = model(test_data) predicted_class = torch.argmax(prediction, dim=1).item() print(f"\nAI Prediction for test data: Class {predicted_class}")
